# Literal colours, type sizes, radii and fills that the design system replaces.
# One list, read by scripts/check.sh (perl) and Tests/VisorCoreTests/DesignAdoptionTests.swift
# (NSRegularExpression), so the check and the test can't disagree. Both scan every Swift
# file under Sources/Visor (Plugins included) line by line, comment lines skipped.
#
# literal<TAB>name<TAB>regex   (Perl and ICU read this syntax the same: lookarounds, \b)
# skip<TAB>path<TAB>why        (a whole file or a folder ending in /; never single lines)
#
# There is no allowlist of lines: move a value into DesignTokens/DesignComponents instead.

literal	hex colour	(NS)?Color\(hex: *(0x|[0-9])
literal	font size	\.system\(size:(?! *(DS\.|CGFloat\((Spec|Piece)\.))
literal	corner radius	cornerRadius:(?! *(DS\.|CGFloat\((Spec|Piece)\.))
literal	opacity fill	opacity\(0?\.[0-9]
literal	white	Color\.white
literal	named colour	(\(|\?\?) *\.(white|black|gray)\b
literal	rgb colour	(?<!NS)Color\((red|white|hue):
literal	AppKit rgb colour	NSColor\((red|srgbRed|white|calibratedRed|deviceRed): *[0-9.]
literal	AppKit font size	ofSize: *[0-9.]
literal	layer corner radius	cornerRadius *= *[0-9.]
literal	alpha component	withAlphaComponent\( *0?\.[0-9]
literal	fill strength	Fill\.white\( *[0-9.]
